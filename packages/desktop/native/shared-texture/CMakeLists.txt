cmake_minimum_required(VERSION 3.20)
project(backspace_shared_texture LANGUAGES CXX)
add_executable(backspace-shared-texture main.cpp)
target_compile_features(backspace-shared-texture PRIVATE cxx_std_17)
target_compile_definitions(backspace-shared-texture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(backspace-shared-texture PRIVATE d3d11 dxgi gdi32 user32)
if(MINGW)
  target_link_options(backspace-shared-texture PRIVATE -static -static-libgcc -static-libstdc++)
endif()
